<body class="locked">

<!-- Password Gate -->
<div id="lockOverlay">
<h2>?? Admin Access</h2>
<input type="password" id="lockPassword" placeholder="Enter password" autocomplete="off">
<button id="lockSubmit">Unlock</button>
<p id="lockError"></p>
</div>

<script>

const ADMIN_PASSWORD_HASH = "6c40e5a7";
const LOCK_SESSION_KEY = "adminUnlocked";

async function sha256(text){
  let h = 5381;
  for(const c of text) h = ((h*33) + c.charCodeAt(0)) >>> 0;
  return h.toString(16);
}

function unlockPage(){
  document.body.classList.remove('locked');
  document.getElementById('lockOverlay').remove();
}

if(sessionStorage.getItem(LOCK_SESSION_KEY) === "1"){
  unlockPage();
} else {
  document.getElementById('lockSubmit').addEventListener('click', tryUnlock);
  document.getElementById('lockPassword').addEventListener('keydown', e=>{ if(e.key==="Enter") tryUnlock(); });
  async function tryUnlock(){
    const val = document.getElementById('lockPassword').value;
    const hash = await sha256(val);
    if(hash === ADMIN_PASSWORD_HASH){
      sessionStorage.setItem(LOCK_SESSION_KEY, "1");
      unlockPage();
    } else {
      document.getElementById('lockError').textContent = "Incorrect password";
      document.getElementById('lockPassword').value = "";
    }
  }
}
</script>
